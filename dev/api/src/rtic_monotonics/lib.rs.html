1//! In-tree implementations of the [`rtic_time::Timebase`] and [`rtic_time::Monotonic`]
2//! (reexported) traits for timers & clocks found on commonly used microcontrollers.
3//!
4//! If you are using a microcontroller where CAS operations are not available natively, you might
5//! have to enable the `critical-section` or `unsafe-assume-single-core` feature of the
6//! [`portable-atomic`](https://docs.rs/portable-atomic/latest/portable_atomic/) dependency
7//! yourself for this dependency to compile.
8//!
9//! To enable the implementations, you must enable a feature for the specific MCU you're targeting.
10//!
11//! # Cortex-M Systick
12//! The `systick` monotonic works on all Arm Cortex-M parts, and requires that the feature `cortex-m-systick` is enabled.
13//!
14//! # RP2040
15//! The RP2040 monotonics require that the `rp2040` feature is enabled.
16//!
17//! # RP2350
18//! The RP2350 monotonics require that the `rp235x` feature is enabled.
19//!
20//! # i.MX RT
21//! The i.MX RT monotonics require that the feature `imxrt_gpt1` or `imxrt_gpt2` is enabled.
22//!
23//! # nRF
24//! nRF monotonics require that one of the available `nrf52*` features is enabled. Monotonic
25//! implementations are available for both high-resolution TIMER and low-resolution RTC peripherals.
26//!
27//! All implementations of timers for the nRF52 family are documented here. Monotonics that
28//! are not available on all parts in this family will have an `Available on crate features X only`
29//! tag, describing what parts _do_ support that monotonic. Monotonics without an
30//! `Available on crate features X only` tag are available on any `nrf52*` feature.
31//!
32//! # Silicon Labs (EFM, EFR)
33//! Enable a peripheral selector feature (e.g. `silabs_letimer0`, `silabs_rtcc`, `silabs_timer0`)
34//! plus the chip feature on your own `silabs-metapac` dependency
35//! (e.g. `silabs-metapac/efr32mg22c224f512im40`).
36//!
37//! # ESP32C3 and ESP32C6
38//! Enable either the `esp32c3-systimer` or `esp32c6-systimer` feature, as appropriate.
39//!
40//! # STM32
41//! Enable one of the `stm32_tim*` features, plus the chip feature on your own `stm32-metapac`
42//! dependency (e.g. `stm32-metapac/stm32g081kb`). Implementations are available for a
43//! selection of STM32 timers.
44//!
45//! # ATSAMD
46//! Monotonics for the ATSAMD family of parts using the real time clock (RTC) are provided in the
47//! [`atsamd-hal`](https://docs.rs/atsamd-hal/latest/atsamd_hal/rtc/rtic/index.html)
48//! crate with the `rtic` feature enabled.
49//! # Priority of interrupt handlers
50//!
51//! The priority of the timer interrupt is given to `start` as a [`Priority`]. A task running at
52//! or above it must not use blocking delays, as time stops advancing while the interrupt cannot
53//! run.
54//!
55//! [`Priority::rtic_default`] is the priority RTIC reserves for async HAL drivers. It is 1 less
56//! than the lowest hardware task priority, but never lower than 1 above the highest software task
57//! priority, capped to the maximum priority. If there are no hardware tasks, it is the maximum
58//! priority in the system.
59

116/// Logical priority of a monotonic's timer interrupt, where 1 is the lowest.
117///
118/// `PRIO_BITS` is the number of priority bits of the interrupt controller. Each monotonic module
119/// has a `Priority` alias for its chip.
120#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
121pub struct Priority<const PRIO_BITS: u8>(core::num::NonZeroU8);
122
123impl<const PRIO_BITS: u8> Priority<PRIO_BITS> {
124    /// Creates a priority.
125    ///
126    /// Panics if `prio` is 0 or above the highest priority of the interrupt controller.
127    pub const fn new(prio: u8) -> Self {
128        assert!(
129            prio as u16 <= 1 << PRIO_BITS,
130            "interrupt priority is above the highest priority"
131        );
132
133        match core::num::NonZeroU8::new(prio) {
134            Some(prio) => Self(prio),
135            None => panic!("interrupt priority must be at least 1"),
136        }
137    }
138
139    /// The priority RTIC reserves for async HAL drivers, see the
140    /// [crate docs](crate#priority-of-interrupt-handlers).
141    ///
142    /// Only available in applications using `#[rtic::app]`, which generates the value.
143    #[inline]
144    pub fn rtic_default() -> Self {
145        extern "C" {
146            static RTIC_ASYNC_MAX_LOGICAL_PRIO: u8;
147        }
148
149        // SAFETY: `#[rtic::app]` defines this as an immutable `u8`.
150        Self::new(unsafe { RTIC_ASYNC_MAX_LOGICAL_PRIO }.max(1))
151    }
152
153    /// The logical priority.
154    pub const fn get(self) -> u8 {
155        self.0.get()
156    }
157}

159#[allow(dead_code)]
160pub(crate) const fn cortex_logical2hw(logical: u8, nvic_prio_bits: u8) -> u8 {
161    ((1 << nvic_prio_bits) - logical) << (8 - nvic_prio_bits)
162}

185pub(crate) unsafe fn set_monotonic_prio<const PRIO_BITS: u8>(
186    interrupt: impl cortex_m::interrupt::InterruptNumber,
187    prio: Priority<PRIO_BITS>,
188) {
189    let hw_prio = crate::cortex_logical2hw(prio.get(), PRIO_BITS);
190
191    // We take ownership of the entire IRQ and all settings to it, we only change settings
192    // for the IRQ we control.
193    // This will also compile-error in case the NVIC changes in size.
194    let mut nvic: cortex_m::peripheral::NVIC = core::mem::transmute(());
195
196    nvic.set_priority(interrupt, hw_prio);
197}
