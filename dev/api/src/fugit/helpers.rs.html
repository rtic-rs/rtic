1/// Needed due to not being allowed to call const-fn in `PartialEq` for some reason
2/// get the error:
3///
4/// ```console
5/// error[E0401]: can't use generic parameters from outer function
6///   --> src/main.rs:25:47
7///    |
8/// 21 | impl<const L_NOM: u32, const L_DENOM: u32, const R_NOM: u32, const R_DENOM: u32>
9///    |                                                                    ------- const parameter from outer function
10/// ...
11/// 25 |         const TEST: u32 = gcd_binary_u32(L_DENOM, R_DENOM);
12///    |                                                   ^^^^^^^ use of generic parameter from outer function
13///
14/// For more information about this error, try `rustc --explain E0401`
15/// ```
16pub struct Helpers<const L_NOM: u64, const L_DENOM: u64, const R_NOM: u64, const R_DENOM: u64>;
17
18impl<const L_NOM: u64, const L_DENOM: u64, const R_NOM: u64, const R_DENOM: u64>
19    Helpers<L_NOM, L_DENOM, R_NOM, R_DENOM>
20{
21    /// Helper constants generated at compile time (intermediate u128 calculation)
22    const DIVISOR_U128: u128 = gcd::binary_u128(
23        L_DENOM as u128 * R_NOM as u128,
24        R_DENOM as u128 * L_NOM as u128,
25    );
26
27    /// Helper constants generated at compile time (intermediate u128 calculation)
28    const DIVISOR_2_U128: u128 = gcd::binary_u128(
29        L_NOM as u128 * R_NOM as u128,
30        R_DENOM as u128 * L_DENOM as u128,
31    );
32
33    /// Helper constants generated at compile time for Durations (intermediate u128 calculation)
34    const RD_TIMES_LN_U128: u128 = (R_DENOM as u128 * L_NOM as u128) / Self::DIVISOR_U128;
35
36    /// Helper constants generated at compile time (intermediate u128 calculation)
37    const LD_TIMES_RN_U128: u128 = (L_DENOM as u128 * R_NOM as u128) / Self::DIVISOR_U128;
38
39    /// Helper constants generated at compile time for Rates (intermediate u128 calculation)
40    const LN_TIMES_RN_U128: u128 = (L_NOM as u128 * R_NOM as u128) / Self::DIVISOR_2_U128;
41
42    /// Helper constants generated at compile time for Rates (intermediate u128 calculation)
43    const RD_TIMES_LD_U128: u128 = (R_DENOM as u128 * L_DENOM as u128) / Self::DIVISOR_2_U128;
44
45    /// Helper constants generated at compile time for Rates (intermediate u128 calculation)
46    const RATE_TO_DURATION_NUMERATOR_U128: u128 = Self::RD_TIMES_LD_U128 / Self::LN_TIMES_RN_U128;
47
48    /// Helper constants generated at compile time for Durations
49    pub const RD_TIMES_LN: u64 = {
50        assert!(
51            Self::RD_TIMES_LN_U128 <= u64::MAX as u128,
52            "RD_TIMES_LN overflows u64 - NOM/DENOM values too large"
53        );
54        Self::RD_TIMES_LN_U128 as u64
55    };
56
57    /// Helper constants generated at compile time
58    pub const LD_TIMES_RN: u64 = {
59        assert!(
60            Self::LD_TIMES_RN_U128 <= u64::MAX as u128,
61            "LD_TIMES_RN overflows u64 - NOM/DENOM values too large"
62        );
63        Self::LD_TIMES_RN_U128 as u64
64    };
65
66    /// Helper constants generated at compile time for Rates
67    pub const RATE_TO_DURATION_NUMERATOR: u64 = {
68        assert!(
69            Self::RATE_TO_DURATION_NUMERATOR_U128 <= u64::MAX as u128,
70            "RATE_TO_DURATION_NUMERATOR overflows u64 - NOM/DENOM values too large"
71        );
72        Self::RATE_TO_DURATION_NUMERATOR_U128 as u64
73    };
74
75    /// Helper constants generated at compile time
76    pub const SAME_BASE: bool = Self::LD_TIMES_RN == Self::RD_TIMES_LN;
77}

79/// Compute `numerator / divisor` rounded to nearest, half-up, without overflow.
80///
81/// Equivalent to `(numerator + divisor / 2) / divisor` when the latter does not
82/// overflow, but stays correct when `numerator` is close to `u64::MAX` (where
83/// adding `divisor / 2` would wrap). `divisor` must be `> 0`.
84///
85/// This is the embedded-friendly form: a single `u64` divmod plus a couple of
86/// cheap 32-bit ops. On 32-bit cores (Cortex-M, RISC-V32) it avoids pulling in
87/// the `u128` libgcc routines that the alternative `(u128 + d/2) / d` shape
88/// would require.
89#[inline]
90pub const fn div_round_nearest_u64(numerator: u64, divisor: u64) -> u64 {
91    let q = numerator / divisor;
92    let r = numerator % divisor;
93    // `r >= d - r` is equivalent to `2 * r >= d` but never overflows because
94    // `r < d`. When d == 1, r is always 0 so the branch is never taken; when
95    // d >= 2, q <= u64::MAX / 2, so `q + 1` cannot overflow either.
96    if r >= divisor - r {
97        q + 1
98    } else {
99        q
100    }
101}

103// Compile-time assert that the cross-base conversion constants fit `$i`. Both are cast
104// `as $i` before use; a constant that fits `u64` but not the storage type would silently
105// truncate, and a divisor truncated to zero traps at runtime. This is the same check the
106// `Duration`/`Rate` shorthands already perform for their fixed bases, extended to the
107// conversions whose bases come from the caller.
108//
109// A same-base conversion reduces both constants to 1, so this never rejects one.
110macro_rules! assert_conversion_fits {
111    ($i:ty, $h:ty, $method:literal) => {
112        const {
113            assert!(
114                <$h>::LD_TIMES_RN <= <$i>::MAX as u64,
115                concat!(
116                    "LD_TIMES_RN doesn't fit in ",
117                    stringify!($i),
118                    " for ",
119                    $method,
120                    " between these time bases"
121                )
122            );
123            assert!(
124                <$h>::RD_TIMES_LN <= <$i>::MAX as u64,
125                concat!(
126                    "RD_TIMES_LN doesn't fit in ",
127                    stringify!($i),
128                    " for ",
129                    $method,
130                    " between these time bases"
131                )
132            );
133        }
134    };
135}
