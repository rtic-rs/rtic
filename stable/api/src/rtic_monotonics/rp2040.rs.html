1//! [`Monotonic`](rtic_time::Monotonic) implementation for RP2040's Timer peripheral.
2//!
3//! Always runs at a fixed rate of 1 MHz.
4//!
5//! # Example
6//!
7//! ```
8//! use rtic_monotonics::rp2040::prelude::*;
9//!
10//! // Create the type `Mono`. It will manage the TIMER peripheral,
11//! // which is a 1 MHz, 64-bit timer.
12//! rp2040_timer_monotonic!(Mono);
13//!
14//! fn init() {
15//!     # // This is normally provided by the selected PAC
16//!     # let TIMER = unsafe { core::mem::transmute(()) };
17//!     # let mut RESETS = unsafe { core::mem::transmute(()) };
18//!     #
19//!     // Start the monotonic - passing ownership of an rp2040_pac object for
20//!     // TIMER0, and temporary access to one for the RESET peripheral.
21//!     Mono::start(TIMER, &mut RESETS, Priority::rtic_default());
22//! }
23//!
24//! async fn usage() {
25//!     loop {
26//!          // You can use the monotonic to get the time...
27//!          let timestamp = Mono::now();
28//!          // ...and you can use it to add a delay to this async function
29//!          Mono::delay(100.millis()).await;
30//!     }
31//! }
32//! ```

51/// Timer implementing [`TimerQueueBackend`].
52pub struct TimerBackend;
53
54impl TimerBackend {
55    /// Starts the monotonic timer.
56    ///
57    /// **Do not use this function directly.**
58    ///
59    /// Use the prelude macros instead.
60    pub fn _start(timer: TIMER, resets: &RESETS, prio: crate::rp2040::Priority) {
61        resets.reset().modify(|_, w| w.timer().clear_bit());
62        while resets.reset_done().read().timer().bit_is_clear() {}
63        timer.inte().modify(|_, w| w.alarm_0().bit(true));
64
65        TIMER_QUEUE.initialize(Self {});
66
67        unsafe {
68            crate::set_monotonic_prio(Interrupt::TIMER_IRQ_0, prio);
69            NVIC::unmask(Interrupt::TIMER_IRQ_0);
70        }
71    }
72
73    fn timer() -> &'static timer::RegisterBlock {
74        unsafe { &*TIMER::ptr() }
75    }
76}
77
78static TIMER_QUEUE: TimerQueue<TimerBackend> = TimerQueue::new();
79
80impl TimerQueueBackend for TimerBackend {
81    type Ticks = u64;
82
83    fn now() -> Self::Ticks {
84        let timer = Self::timer();
85
86        let mut hi0 = timer.timerawh().read().bits();
87        loop {
88            let low = timer.timerawl().read().bits();
89            let hi1 = timer.timerawh().read().bits();
90            if hi0 == hi1 {
91                break (u64::from(hi0) << 32) | u64::from(low);
92            }
93            hi0 = hi1;
94        }
95    }
96
97    fn set_compare(instant: Self::Ticks) {
98        let now = Self::now();
99
100        const MAX: u64 = u32::MAX as u64;
101
102        // Since the timer may or may not overflow based on the requested compare val, we check
103        // how many ticks are left.
104        // `wrapping_sub` takes care of the u64 integer overflow special case.
105        let val = if instant.wrapping_sub(now) <= MAX {
106            instant & MAX
107        } else {
108            0
109        };
110
111        Self::timer()
112            .alarm0()
113            .write(|w| unsafe { w.bits(val as u32) });
114    }
115
116    fn clear_compare_flag() {
117        Self::timer().intr().modify(|_, w| w.alarm_0().bit(true));
118    }
119
120    fn pend_interrupt() {
121        rp2040_pac::NVIC::pend(Interrupt::TIMER_IRQ_0);
122    }
123
124    fn timer_queue() -> &'static TimerQueue<Self> {
125        &TIMER_QUEUE
126    }
127}

129/// Create an RP2040 timer based monotonic and register the necessary interrupt for it.
130///
131/// See [`crate::rp2040`] for more details.
132///
133/// # Arguments
134///
135/// * `name` - The name that the monotonic type will have.
136#[macro_export]
137macro_rules! rp2040_timer_monotonic {
138    ($name:ident) => {
139        /// A `Monotonic` based on the RP2040 Timer peripheral.
140        pub struct $name;
141
142        impl $name {
143            /// Starts the `Monotonic`.
144            ///
145            /// `prio` is the priority of the timer interrupt.
146            ///
147            /// This method must be called only once.
148            pub fn start(
149                timer: $crate::rp2040::TIMER,
150                resets: &$crate::rp2040::RESETS,
151                prio: $crate::rp2040::Priority,
152            ) {
153                #[no_mangle]
154                #[allow(non_snake_case)]
155                unsafe extern "C" fn TIMER_IRQ_0() {
156                    use $crate::TimerQueueBackend;
157                    $crate::rp2040::TimerBackend::timer_queue().on_monotonic_interrupt();
158                }
159
160                $crate::rp2040::TimerBackend::_start(timer, resets, prio);
161            }
162        }
163
164        impl $crate::TimerQueueBasedMonotonic for $name {
165            type Backend = $crate::rp2040::TimerBackend;
166            type Instant = $crate::fugit::MonotonicTimerInstant<
167                <Self::Backend as $crate::TimerQueueBackend>::Ticks,
168                1_000_000,
169            >;
170            type Duration = $crate::fugit::TimerDuration<
171                <Self::Backend as $crate::TimerQueueBackend>::Ticks,
172                1_000_000,
173            >;
174        }
175
176        $crate::rtic_time::impl_embedded_hal_delay_fugit!($name);
177        $crate::rtic_time::impl_embedded_hal_async_delay_fugit!($name);
178    };
179}
