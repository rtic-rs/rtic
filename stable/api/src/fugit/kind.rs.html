1//! Markers that say how an [`Instant`](crate::Instant)'s timeline is read.
2
3mod sealed {
4    pub trait Sealed {}
5}
6
7/// The set of timelines an [`Instant`](crate::Instant) can be on.
8///
9/// Sealed: the kinds are [`Monotonic`] and [`Wrapping`].
10pub trait Kind: Copy + sealed::Sealed {
11    /// The name shown for instants of this kind in `Debug` output.
12    const DEBUG_NAME: &'static str;
13}
14
15/// Marker for an instant on a timeline that does not wrap.
16///
17/// Total order, so [`Ord`]. Arithmetic is plain `+` and `-`, panicking on overflow in debug and
18/// wrapping in release. Nothing checks that the timeline really does not wrap, so use
19/// [`Wrapping`] for raw hardware counters.
20#[derive(Copy, Clone, Hash, PartialEq, Eq)]
21pub enum Monotonic {}
22
23impl sealed::Sealed for Monotonic {}
24impl Kind for Monotonic {
25    const DEBUG_NAME: &'static str = "MonotonicInstant";
26}
27
28/// Marker for an instant on a circular counter, such as a hardware timer.
29///
30/// Arithmetic wraps, and comparison reads `self - other` as a signed offset, so it is only
31/// meaningful within half the tick range and not transitive, which is less than [`PartialOrd`]
32/// and [`Ord`] promise, so neither is implemented. Comparison goes through
33/// [`InstantOrd`](crate::InstantOrd) or the inherent `is_before`/`is_after`/`const_partial_cmp`:
34///
35/// ```compile_fail
36/// use fugit::WrappingInstant;
37///
38/// fn assert_partial_ord<T: PartialOrd>() {}
39/// assert_partial_ord::<WrappingInstant<u32, 1, 1_000>>();
40/// ```
41#[derive(Copy, Clone, Hash, PartialEq, Eq)]
42pub enum Wrapping {}
43
44impl sealed::Sealed for Wrapping {}
45impl Kind for Wrapping {
46    const DEBUG_NAME: &'static str = "WrappingInstant";
47}
