8/// Represents an instant in time.
9///
10/// The generic `T` can either be `u32` or `u64`, and the const generics represent the ratio of the
11/// ticks contained within the instant: `instant in seconds = NOM / DENOM * ticks`
12///
13/// `K` says how the timeline is read. See [`kind`](crate::kind), or use an alias such as
14/// [`WrappingInstant`](crate::WrappingInstant).
15///
16/// Adding or subtracting a [`Duration`] in a different time base is subject to the same base
17/// ratio limits as [`Duration`] itself: bases differing by more than `T::MAX` are a compile-time
18/// error rather than a silent truncation.
19#[cfg_attr(feature = "serde", derive(serde::Deserialize, serde::Serialize))]
20#[derive(Clone, Copy, Hash, PartialEq, Eq)]
21pub struct Instant<T, const NOM: u64, const DENOM: u64, K> {
22    ticks: T,
23    #[cfg_attr(feature = "serde", serde(skip))]
24    _kind: PhantomData<K>,
25}

29    postcard::experimental::max_size::MaxSize for Instant<T, NOM, DENOM, K>
30{
31    const POSTCARD_MAX_SIZE: usize = T::POSTCARD_MAX_SIZE;
32}

44/// Ordering between two instants on the same timeline.
45///
46/// The comparison interface shared by both instant kinds: a total order on
47/// [`Monotonic`](crate::kind::Monotonic) instants, wrap-aware on
48/// [`Wrapping`](crate::kind::Wrapping) instants. Bound on [`Ord`] instead where a total order
49/// is required.
50pub trait InstantOrd: Copy {
51    /// Compare two instants, `None` when they have no order.
52    fn compare(self, other: Self) -> Option<Ordering>;
53
54    /// Whether `self` comes before `other`, `false` if they have no order.
55    fn is_before(self, other: Self) -> bool {
56        matches!(self.compare(other), Some(Ordering::Less))
57    }
58
59    /// Whether `self` comes after `other`, `false` if they have no order.
60    fn is_after(self, other: Self) -> bool {
61        matches!(self.compare(other), Some(Ordering::Greater))
62    }
63
64    /// Whether `self` is at the same time as `other` or later (a >= b filler), `false` if
65    /// they have no order.
66    fn is_at_least(self, other: Self) -> bool {
67        matches!(
68            self.compare(other),
69            Some(Ordering::Equal) | Some(Ordering::Greater)
70        )
71    }
72}

159/// Comparison and arithmetic for instants read as a circular counter: ticks wrap, and the
160/// wrap-aware ordering is exposed through [`InstantOrd`] and the inherent methods.
161macro_rules! impl_instant_wrapping {
162    ($i:ty) => {
163        impl<const NOM: u64, const DENOM: u64> Instant<$i, NOM, DENOM, Wrapping> {
164            /// Const partial comparison of `Instant`s.
165            ///
166            /// ```
167            /// # use fugit::*;
168            #[doc = concat!("let i1 = WrappingInstant::<", stringify!($i), ", 1, 1_000>::from_ticks(1);")]
169            #[doc = concat!("let i2 = WrappingInstant::<", stringify!($i), ", 1, 1_000>::from_ticks(2);")]
170            ///
171            /// assert_eq!(i1.const_partial_cmp(i2), Some(core::cmp::Ordering::Less));
172            /// ```
173            ///
174            /// This function takes into account that ticks might wrap around: `self - other` is
175            /// read as a signed offset, so an `Instant` just past the wrap point comes after one
176            /// just before it:
177            ///
178            /// ```
179            /// # use fugit::*;
180            #[doc = concat!("let i1 = WrappingInstant::<", stringify!($i), ", 1, 1_000>::from_ticks(", stringify!($i),"::MAX);")]
181            #[doc = concat!("let i2 = WrappingInstant::<", stringify!($i), ", 1, 1_000>::from_ticks(1);")]
182            ///
183            /// assert_eq!(i1.const_partial_cmp(i2), Some(core::cmp::Ordering::Less));
184            /// ```
185            ///
186            /// Returns `None` when the two are exactly half the tick range apart. There
187            /// `self - other` and `other - self` are the same value, so neither can be said to
188            /// come first:
189            ///
190            /// ```
191            /// # use fugit::*;
192            #[doc = concat!("let i1 = WrappingInstant::<", stringify!($i), ", 1, 1_000>::from_ticks(0);")]
193            #[doc = concat!("let i2 = WrappingInstant::<", stringify!($i), ", 1, 1_000>::from_ticks(1 << (", stringify!($i), "::BITS - 1));")]
194            ///
195            /// assert_eq!(i1.const_partial_cmp(i2), None);
196            /// ```
197            #[inline]
198            pub const fn const_partial_cmp(self, other: Self) -> Option<Ordering> {
199                const HALF: $i = <$i>::MAX / 2 + 1;
200
201                let v = self.ticks.wrapping_sub(other.ticks);
202
203                // not using `cmp` due to it being non-const
204                if v == 0 {
205                    Some(Ordering::Equal)
206                } else if v < HALF {
207                    Some(Ordering::Greater)
208                } else if v > HALF {
209                    Some(Ordering::Less)
210                } else {
211                    None
212                }
213            }
214
215            /// Whether this `Instant` comes before `other`.
216            ///
217            /// Wrap-aware, like [`const_partial_cmp`](Self::const_partial_cmp), and usable in
218            /// const contexts. Instants exactly half the tick range apart are
219            /// incomparable, so this and [`is_after`](Self::is_after) are both `false` for them.
220            ///
221            /// ```
222            /// # use fugit::*;
223            #[doc = concat!("const I1: WrappingInstant<", stringify!($i), ", 1, 1_000>")]
224            #[doc = concat!("    = WrappingInstant::<", stringify!($i), ", 1, 1_000>::from_ticks(", stringify!($i), "::MAX);")]
225            #[doc = concat!("const I2: WrappingInstant<", stringify!($i), ", 1, 1_000>")]
226            #[doc = concat!("    = WrappingInstant::<", stringify!($i), ", 1, 1_000>::from_ticks(1);")]
227            /// const BEFORE: bool = I1.is_before(I2);
228            ///
229            /// assert!(BEFORE);
230            /// ```
231            #[inline]
232            pub const fn is_before(self, other: Self) -> bool {
233                matches!(self.const_partial_cmp(other), Some(Ordering::Less))
234            }
235
236            /// Whether this `Instant` comes after `other`.
237            ///
238            /// Wrap-aware, like [`const_partial_cmp`](Self::const_partial_cmp), and usable in
239            /// const contexts. Instants exactly half the tick range apart are
240            /// incomparable, so this and [`is_before`](Self::is_before) are both `false` for them.
241            ///
242            /// ```
243            /// # use fugit::*;
244            #[doc = concat!("const I1: WrappingInstant<", stringify!($i), ", 1, 1_000>")]
245            #[doc = concat!("    = WrappingInstant::<", stringify!($i), ", 1, 1_000>::from_ticks(1);")]
246            #[doc = concat!("const I2: WrappingInstant<", stringify!($i), ", 1, 1_000>")]
247            #[doc = concat!("    = WrappingInstant::<", stringify!($i), ", 1, 1_000>::from_ticks(", stringify!($i), "::MAX);")]
248            /// const AFTER: bool = I1.is_after(I2);
249            ///
250            /// assert!(AFTER);
251            /// ```
252            #[inline]
253            pub const fn is_after(self, other: Self) -> bool {
254                matches!(self.const_partial_cmp(other), Some(Ordering::Greater))
255            }
256
257            /// Duration between `Instant`s.
258            ///
259            /// Returns `None` if `self` is before `other`, or if the two cannot be
260            /// ordered at all, under the wrap-aware ordering used by
261            /// [`const_partial_cmp`](Self::const_partial_cmp).
262            ///
263            /// ```
264            /// # use fugit::*;
265            #[doc = concat!("let i1 = WrappingInstant::<", stringify!($i), ", 1, 1_000>::from_ticks(1);")]
266            #[doc = concat!("let i2 = WrappingInstant::<", stringify!($i), ", 1, 1_000>::from_ticks(2);")]
267            ///
268            /// assert_eq!(i1.checked_duration_since(i2), None);
269            /// assert_eq!(i2.checked_duration_since(i1).unwrap().as_ticks(), 1);
270            /// ```
271            #[inline]
272            pub const fn checked_duration_since(
273                self,
274                other: Self,
275            ) -> Option<Duration<$i, NOM, DENOM>> {
276                match self.const_partial_cmp(other) {
277                    Some(Ordering::Greater) | Some(Ordering::Equal) => {
278                        Some(Duration::<$i, NOM, DENOM>::from_ticks(
279                            self.ticks.wrapping_sub(other.ticks),
280                        ))
281                    }
282                    Some(Ordering::Less) | None => None,
283                }
284            }
285
286            /// Try to convert `other` into the `Self::NOM / Self::DENOM` timebase, and
287            /// subtract it from this [`Instant`].
288            ///
289            /// Returns `None` only if the time-base conversion fails. The subtraction itself
290            /// is wrapping, as [`Instant`]s are circular.
291            ///
292            /// The implementations of [`core::ops::Sub`] and [`core::ops::SubAssign`] are
293            /// implemented by `unwrap`-ing the value returned by this function.
294            ///
295            /// ```
296            /// # use fugit::*;
297            #[doc = concat!("let i = WrappingInstant::<", stringify!($i), ", 1, 1_000>::from_ticks(1);")]
298            #[doc = concat!("let d = Duration::<", stringify!($i), ", 1, 1_000>::from_ticks(1);")]
299            ///
300            /// assert_eq!(i.convert_sub_duration(d).unwrap().as_ticks(), 0);
301            /// ```
302            ///
303            /// The subtraction itself is wrapping:
304            ///
305            /// ```
306            /// # use fugit::*;
307            #[doc = concat!("let i = WrappingInstant::<", stringify!($i), ", 1, 1_000>::from_ticks(1);")]
308            #[doc = concat!("let d = Duration::<", stringify!($i), ", 1, 1_000>::from_ticks(2);")]
309            ///
310            #[doc = concat!("assert_eq!(i.convert_sub_duration(d).unwrap().as_ticks(), ", stringify!($i), "::MAX);")]
311            /// ```
312            ///
313            /// Overflow during [`Duration`] base conversion returns `None`:
314            ///
315            /// ```
316            /// # use fugit::*;
317            #[doc = concat!("let i = WrappingInstant::<", stringify!($i), ", 1, 1_000>::from_ticks(1);")]
318            /// // A duration whose conversion from the `1/500` base to the `1/1000` base
319            /// // will overflow.
320            #[doc = concat!("let d = Duration::<", stringify!($i), ", 1, 500>::from_ticks(", stringify!($i),"::MAX / 2 + 1);")]
321            ///
322            /// assert_eq!(i.convert_sub_duration(d), None);
323            /// ```
324            pub const fn convert_sub_duration<const O_NOM: u64, const O_DENOM: u64>(
325                self,
326                other: Duration<$i, O_NOM, O_DENOM>,
327            ) -> Option<Self> {
328                assert_conversion_fits!(
329                    $i,
330                    Helpers::<NOM, DENOM, O_NOM, O_DENOM>,
331                    "Instant::convert_sub_duration"
332                );
333
334                if Helpers::<NOM, DENOM, O_NOM, O_DENOM>::SAME_BASE {
335                    Some(Self::from_ticks(
336                        self.ticks.wrapping_sub(other.as_ticks()),
337                    ))
338                } else {
339                    if let Some(lh) = other
340                        .as_ticks()
341                        .checked_mul(Helpers::<NOM, DENOM, O_NOM, O_DENOM>::LD_TIMES_RN as $i)
342                    {
343                        let ticks = lh / Helpers::<NOM, DENOM, O_NOM, O_DENOM>::RD_TIMES_LN as $i;
344
345                        Some(Self::from_ticks(self.ticks.wrapping_sub(ticks)))
346                    } else {
347                        None
348                    }
349                }
350            }
351
352            /// Try to convert `other` into the `Self::NOM / Self::DENOM` timebase, and
353            /// add it to this [`Instant`].
354            ///
355            /// Returns `None` only if the time-base conversion fails. The addition itself
356            /// is wrapping, as [`Instant`]s are circular.
357            ///
358            /// The implementations of [`core::ops::Add`] and [`core::ops::AddAssign`] are
359            /// implemented by `unwrap`-ing the value returned by this function.
360            ///
361            /// ```
362            /// # use fugit::*;
363            #[doc = concat!("let i = WrappingInstant::<", stringify!($i), ", 1, 1_000>::from_ticks(1);")]
364            #[doc = concat!("let d = Duration::<", stringify!($i), ", 1, 1_000>::from_ticks(1);")]
365            ///
366            /// assert_eq!(i.convert_add_duration(d).unwrap().as_ticks(), 2);
367            /// ```
368            ///
369            /// The addition itself is wrapping:
370            ///
371            /// ```
372            /// # use fugit::*;
373            #[doc = concat!("let i = WrappingInstant::<", stringify!($i), ", 1, 1_000>::from_ticks(1);")]
374            #[doc = concat!("let d = Duration::<", stringify!($i), ", 1, 1_000>::from_ticks(", stringify!($i), "::MAX);")]
375            ///
376            /// assert_eq!(i.convert_add_duration(d).unwrap().as_ticks(), 0);
377            /// ```
378            ///
379            /// Overflow during [`Duration`] base conversion returns `None`:
380            ///
381            /// ```
382            /// # use fugit::*;
383            #[doc = concat!("let i = WrappingInstant::<", stringify!($i), ", 1, 1_000>::from_ticks(1);")]
384            /// // A duration whose conversion from the `1/500` base to the `1/1000` base
385            /// // will overflow.
386            #[doc = concat!("let d = Duration::<", stringify!($i), ", 1, 500>::from_ticks(", stringify!($i),"::MAX / 2 + 1);")]
387            ///
388            /// assert_eq!(i.convert_add_duration(d), None);
389            /// ```
390            pub const fn convert_add_duration<const O_NOM: u64, const O_DENOM: u64>(
391                self,
392                other: Duration<$i, O_NOM, O_DENOM>,
393            ) -> Option<Self> {
394                assert_conversion_fits!(
395                    $i,
396                    Helpers::<NOM, DENOM, O_NOM, O_DENOM>,
397                    "Instant::convert_add_duration"
398                );
399
400                if Helpers::<NOM, DENOM, O_NOM, O_DENOM>::SAME_BASE {
401                    Some(Self::from_ticks(
402                        self.ticks.wrapping_add(other.as_ticks()),
403                    ))
404                } else {
405                    if let Some(lh) = other
406                        .as_ticks()
407                        .checked_mul(Helpers::<NOM, DENOM, O_NOM, O_DENOM>::LD_TIMES_RN as $i)
408                    {
409                        let ticks = lh / Helpers::<NOM, DENOM, O_NOM, O_DENOM>::RD_TIMES_LN as $i;
410
411                        Some(Self::from_ticks(self.ticks.wrapping_add(ticks)))
412                    } else {
413                        None
414                    }
415                }
416            }
417        }
418
419        impl<const NOM: u64, const DENOM: u64> InstantOrd for Instant<$i, NOM, DENOM, Wrapping> {
420            /// Wrap-aware, see [`const_partial_cmp`](Instant::const_partial_cmp).
421            #[inline]
422            fn compare(self, other: Self) -> Option<Ordering> {
423                self.const_partial_cmp(other)
424            }
425        }
426
427        // Instant - Duration = Instant
428        // We have limited this to use same numerator and denominator in both left and right hand sides,
429        // this allows for the extension traits to work. For usage with different fraction, use
430        // [`Self::convert_sub_duration`].
431        impl<const NOM: u64, const DENOM: u64> ops::Sub<Duration<$i, NOM, DENOM>>
432            for Instant<$i, NOM, DENOM, Wrapping>
433        {
434            type Output = Self;
435
436            #[inline]
437            #[track_caller]
438            fn sub(self, other: Duration<$i, NOM, DENOM>) -> Self::Output {
439                if let Some(v) = self.convert_sub_duration(other) {
440                    v
441                } else {
442                    panic!("Sub failed! Overflow");
443                }
444            }
445        }
446
447        // Instant + Duration = Instant
448        // We have limited this to use same numerator and denominator in both left and right hand sides,
449        // this allows for the extension traits to work. For usage with different fraction, use
450        // [`Self::convert_add_duration`].
451        impl<const NOM: u64, const DENOM: u64> ops::Add<Duration<$i, NOM, DENOM>>
452            for Instant<$i, NOM, DENOM, Wrapping>
453        {
454            type Output = Self;
455
456            #[inline]
457            #[track_caller]
458            fn add(self, other: Duration<$i, NOM, DENOM>) -> Self::Output {
459                if let Some(v) = self.convert_add_duration(other) {
460                    v
461                } else {
462                    panic!("Add failed! Overflow");
463                }
464            }
465        }
466    };

469/// Comparison and arithmetic for instants on a timeline that does not wrap: tick math is plain
470/// integer math, and the ordering is total.
471macro_rules! impl_instant_monotonic {
472    ($i:ty) => {
473        impl<const NOM: u64, const DENOM: u64> Instant<$i, NOM, DENOM, Monotonic> {
474            /// Const comparison of `Instant`s.
475            ///
476            /// ```
477            /// # use fugit::*;
478            #[doc = concat!("let i1 = MonotonicInstant::<", stringify!($i), ", 1, 1_000>::from_ticks(1);")]
479            #[doc = concat!("let i2 = MonotonicInstant::<", stringify!($i), ", 1, 1_000>::from_ticks(2);")]
480            ///
481            /// assert_eq!(i1.const_cmp(i2), core::cmp::Ordering::Less);
482            /// ```
483            #[inline]
484            pub const fn const_cmp(self, other: Self) -> Ordering {
485                // not using `cmp` due to it being non-const
486                if self.ticks < other.ticks {
487                    Ordering::Less
488                } else if self.ticks > other.ticks {
489                    Ordering::Greater
490                } else {
491                    Ordering::Equal
492                }
493            }
494
495            /// Whether this `Instant` comes before `other`.
496            ///
497            /// Usable in const contexts, where `<` is not.
498            ///
499            /// ```
500            /// # use fugit::*;
501            #[doc = concat!("const I1: MonotonicInstant<", stringify!($i), ", 1, 1_000>")]
502            #[doc = concat!("    = MonotonicInstant::<", stringify!($i), ", 1, 1_000>::from_ticks(1);")]
503            #[doc = concat!("const I2: MonotonicInstant<", stringify!($i), ", 1, 1_000>")]
504            #[doc = concat!("    = MonotonicInstant::<", stringify!($i), ", 1, 1_000>::from_ticks(2);")]
505            /// const BEFORE: bool = I1.is_before(I2);
506            ///
507            /// assert!(BEFORE);
508            /// ```
509            #[inline]
510            pub const fn is_before(self, other: Self) -> bool {
511                self.ticks < other.ticks
512            }
513
514            /// Whether this `Instant` comes after `other`.
515            ///
516            /// Usable in const contexts, where `>` is not.
517            ///
518            /// ```
519            /// # use fugit::*;
520            #[doc = concat!("const I1: MonotonicInstant<", stringify!($i), ", 1, 1_000>")]
521            #[doc = concat!("    = MonotonicInstant::<", stringify!($i), ", 1, 1_000>::from_ticks(2);")]
522            #[doc = concat!("const I2: MonotonicInstant<", stringify!($i), ", 1, 1_000>")]
523            #[doc = concat!("    = MonotonicInstant::<", stringify!($i), ", 1, 1_000>::from_ticks(1);")]
524            /// const AFTER: bool = I1.is_after(I2);
525            ///
526            /// assert!(AFTER);
527            /// ```
528            #[inline]
529            pub const fn is_after(self, other: Self) -> bool {
530                self.ticks > other.ticks
531            }
532
533            /// The duration between this instant and `Instant::from_ticks(0)`.
534            ///
535            /// ```
536            /// # use fugit::*;
537            #[doc = concat!("let i = MonotonicInstant::<", stringify!($i), ", 1, 1_000>::from_ticks(11);")]
538            ///
539            /// assert_eq!(i.duration_since_epoch().as_ticks(), 11);
540            /// ```
541            #[inline]
542            pub const fn duration_since_epoch(self) -> Duration<$i, NOM, DENOM> {
543                Duration::<$i, NOM, DENOM>::from_ticks(self.as_ticks())
544            }
545
546            /// Duration between `Instant`s.
547            ///
548            /// Returns `None` if `self` is before `other`.
549            ///
550            /// ```
551            /// # use fugit::*;
552            #[doc = concat!("let i1 = MonotonicInstant::<", stringify!($i), ", 1, 1_000>::from_ticks(1);")]
553            #[doc = concat!("let i2 = MonotonicInstant::<", stringify!($i), ", 1, 1_000>::from_ticks(2);")]
554            ///
555            /// assert_eq!(i1.checked_duration_since(i2), None);
556            /// assert_eq!(i2.checked_duration_since(i1).unwrap().as_ticks(), 1);
557            /// ```
558            #[inline]
559            pub const fn checked_duration_since(
560                self,
561                other: Self,
562            ) -> Option<Duration<$i, NOM, DENOM>> {
563                match self.ticks.checked_sub(other.ticks) {
564                    Some(ticks) => Some(Duration::<$i, NOM, DENOM>::from_ticks(ticks)),
565                    None => None,
566                }
567            }
568
569            /// Try to convert `other` into the `Self::NOM / Self::DENOM` timebase, and
570            /// subtract it from this [`Instant`].
571            ///
572            /// Returns `None` if the time-base conversion fails, or if the subtraction would go
573            /// below zero. Note that those two causes are not distinguished.
574            ///
575            /// ```
576            /// # use fugit::*;
577            #[doc = concat!("let i = MonotonicInstant::<", stringify!($i), ", 1, 1_000>::from_ticks(1);")]
578            #[doc = concat!("let d = Duration::<", stringify!($i), ", 1, 1_000>::from_ticks(1);")]
579            ///
580            /// assert_eq!(i.checked_sub_duration(d).unwrap().as_ticks(), 0);
581            /// ```
582            ///
583            /// Going below zero returns `None` rather than wrapping:
584            ///
585            /// ```
586            /// # use fugit::*;
587            #[doc = concat!("let i = MonotonicInstant::<", stringify!($i), ", 1, 1_000>::from_ticks(1);")]
588            #[doc = concat!("let d = Duration::<", stringify!($i), ", 1, 1_000>::from_ticks(2);")]
589            ///
590            /// assert_eq!(i.checked_sub_duration(d), None);
591            /// ```
592            pub const fn checked_sub_duration<const O_NOM: u64, const O_DENOM: u64>(
593                self,
594                other: Duration<$i, O_NOM, O_DENOM>,
595            ) -> Option<Self> {
596                assert_conversion_fits!(
597                    $i,
598                    Helpers::<NOM, DENOM, O_NOM, O_DENOM>,
599                    "Instant::checked_sub_duration"
600                );
601
602                match Duration::<$i, NOM, DENOM>::from_ticks(self.ticks).checked_sub(other) {
603                    Some(d) => Some(Self::from_ticks(d.as_ticks())),
604                    None => None,
605                }
606            }
607
608            /// Try to convert `other` into the `Self::NOM / Self::DENOM` timebase, and
609            /// add it to this [`Instant`].
610            ///
611            /// Returns `None` if the time-base conversion fails, or if the addition would
612            /// overflow the storage type. Note that those two causes are not distinguished.
613            ///
614            /// ```
615            /// # use fugit::*;
616            #[doc = concat!("let i = MonotonicInstant::<", stringify!($i), ", 1, 1_000>::from_ticks(1);")]
617            #[doc = concat!("let d = Duration::<", stringify!($i), ", 1, 1_000>::from_ticks(1);")]
618            ///
619            /// assert_eq!(i.checked_add_duration(d).unwrap().as_ticks(), 2);
620            /// ```
621            ///
622            /// Overflowing the storage type returns `None` rather than wrapping:
623            ///
624            /// ```
625            /// # use fugit::*;
626            #[doc = concat!("let i = MonotonicInstant::<", stringify!($i), ", 1, 1_000>::from_ticks(1);")]
627            #[doc = concat!("let d = Duration::<", stringify!($i), ", 1, 1_000>::from_ticks(", stringify!($i), "::MAX);")]
628            ///
629            /// assert_eq!(i.checked_add_duration(d), None);
630            /// ```
631            pub const fn checked_add_duration<const O_NOM: u64, const O_DENOM: u64>(
632                self,
633                other: Duration<$i, O_NOM, O_DENOM>,
634            ) -> Option<Self> {
635                assert_conversion_fits!(
636                    $i,
637                    Helpers::<NOM, DENOM, O_NOM, O_DENOM>,
638                    "Instant::checked_add_duration"
639                );
640
641                match Duration::<$i, NOM, DENOM>::from_ticks(self.ticks).checked_add(other) {
642                    Some(d) => Some(Self::from_ticks(d.as_ticks())),
643                    None => None,
644                }
645            }
646        }
647
648        impl<const NOM: u64, const DENOM: u64> Ord for Instant<$i, NOM, DENOM, Monotonic> {
649            #[inline]
650            fn cmp(&self, other: &Self) -> Ordering {
651                self.ticks.cmp(&other.ticks)
652            }
653        }
654
655        impl<const NOM: u64, const DENOM: u64> PartialOrd for Instant<$i, NOM, DENOM, Monotonic> {
656            #[inline]
657            fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
658                Some(self.cmp(other))
659            }
660        }
661
662        impl<const NOM: u64, const DENOM: u64> InstantOrd for Instant<$i, NOM, DENOM, Monotonic> {
663            /// Always `Some`, the timeline is totally ordered.
664            #[inline]
665            fn compare(self, other: Self) -> Option<Ordering> {
666                Some(self.const_cmp(other))
667            }
668        }

670        // Instant - Duration = Instant
671        // We have limited this to use same numerator and denominator in both left and right hand sides,
672        // this allows for the extension traits to work. For usage with different fraction, use
673        // [`Self::checked_sub_duration`].
674        //
675        // Plain `-`, so this inherits `overflow-checks` from the final binary: a panic in debug and
676        // a wrap in release, exactly like the underlying integer.
677        impl<const NOM: u64, const DENOM: u64> ops::Sub<Duration<$i, NOM, DENOM>>
678            for Instant<$i, NOM, DENOM, Monotonic>
679        {
680            type Output = Self;
681
682            #[inline]
683            #[track_caller]
684            fn sub(self, other: Duration<$i, NOM, DENOM>) -> Self::Output {
685                Self::from_ticks(self.ticks - other.as_ticks())
686            }
687        }
688
689        // Instant + Duration = Instant
690        // We have limited this to use same numerator and denominator in both left and right hand sides,
691        // this allows for the extension traits to work. For usage with different fraction, use
692        // [`Self::checked_add_duration`].
693        //
694        // Plain `+`, see the `Sub` impl above.
695        impl<const NOM: u64, const DENOM: u64> ops::Add<Duration<$i, NOM, DENOM>>
696            for Instant<$i, NOM, DENOM, Monotonic>
697        {
698            type Output = Self;
699
700            #[inline]
701            #[track_caller]
702            fn add(self, other: Duration<$i, NOM, DENOM>) -> Self::Output {
703                Self::from_ticks(self.ticks + other.as_ticks())
704            }
705        }
706    };
707}

709/// `Instant - Instant = Duration`, the one operator body both kinds share.
710macro_rules! impl_instant_ops {
711    ($i:ty, $k:ty) => {
712        // Instant - Instant = Duration
713        // We have limited this to use same numerator and denominator in both left and right hand sides,
714        // this allows for the extension traits to work. For usage with different fraction, use
715        // `checked_duration_since`.
716        impl<const NOM: u64, const DENOM: u64> ops::Sub<Instant<$i, NOM, DENOM, $k>>
717            for Instant<$i, NOM, DENOM, $k>
718        {
719            type Output = Duration<$i, NOM, DENOM>;
720
721            #[inline]
722            #[track_caller]
723            fn sub(self, other: Self) -> Self::Output {
724                if let Some(v) = self.checked_duration_since(other) {
725                    v
726                } else {
727                    panic!("Sub failed! Other is not before self");
728                }
729            }
730        }
731    };
732}
