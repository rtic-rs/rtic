7/// Represents a frequency.
8///
9/// The generic `T` can either be `u32` or `u64`, and the const generics represent the ratio of the
10/// raw contained within the rate: `rate in Hz = NOM / DENOM * raw`
11///
12/// Operations between two different bases are subject to the same base ratio limits as
13/// [`Duration`]: bases differing by more than `T::MAX` are a compile-time error rather than a
14/// silent truncation.
15#[cfg_attr(feature = "serde", derive(serde::Deserialize, serde::Serialize))]
16#[cfg_attr(
17    feature = "postcard_max_size",
18    derive(postcard::experimental::max_size::MaxSize)
19)]
20#[derive(Clone, Copy, Debug)]
21pub struct Rate<T, const NOM: u64, const DENOM: u64> {
22    pub(crate) raw: T,
23}

25// Unwrap a const-time `Option` or panic with the given static message. Lets the
26// shorthand methods turn `checked_mul`/`checked_add` failures into a clear panic
27// at the user's call site instead of silently wrapping in release builds.
28macro_rules! const_checked {
29    ($e:expr, $msg:expr) => {
30        match $e {
31            Some(v) => v,
32            None => panic!("{}", $msg),
33        }
34    };
35}

37// Compile-time assert that the helper constants for (l_nom, l_denom, NOM, DENOM)
38// fit in `$i`. The Hz/kHz/MHz/to_Hz/to_kHz/to_MHz shorthands cast both constants
39// `as $i`; without this check those casts silently truncate when the value fits
40// `u64` but not the target storage type.
41macro_rules! assert_helpers_fit {
42    ($i:ty, $l_nom:literal, $l_denom:literal, $method:literal) => {
43        const {
44            assert!(
45                Helpers::<$l_nom, $l_denom, NOM, DENOM>::LD_TIMES_RN <= <$i>::MAX as u64,
46                concat!(
47                    "LD_TIMES_RN doesn't fit in ",
48                    stringify!($i),
49                    " for ",
50                    $method,
51                    " on this Rate type"
52                )
53            );
54            assert!(
55                Helpers::<$l_nom, $l_denom, NOM, DENOM>::RD_TIMES_LN <= <$i>::MAX as u64,
56                concat!(
57                    "RD_TIMES_LN doesn't fit in ",
58                    stringify!($i),
59                    " for ",
60                    $method,
61                    " on this Rate type"
62                )
63            );
64        }
65    };
66}

98            /// Add two rates.
99            ///
100            /// Returns `None` on raw overflow or cross-base conversion overflow.
101            ///
102            /// ```
103            /// # use fugit::*;
104            #[doc = concat!("let r1 = Rate::<", stringify!($i), ", 1, 1_000>::from_raw(1);")]
105            #[doc = concat!("let r2 = Rate::<", stringify!($i), ", 1, 1_000>::from_raw(2);")]
106            #[doc = concat!("let r3 = Rate::<", stringify!($i), ", 1, 1_000>::from_raw(", stringify!($i), "::MAX);")]
107            ///
108            /// assert_eq!(r1.checked_add(r2).unwrap().to_raw(), 3);
109            /// assert_eq!(r1.checked_add(r3), None);
110            /// ```
111            pub const fn checked_add<const O_NOM: u64, const O_DENOM: u64>(
112                self,
113                other: Rate<$i, O_NOM, O_DENOM>,
114            ) -> Option<Self> {
115                assert_conversion_fits!(
116                    $i,
117                    Helpers::<NOM, DENOM, O_NOM, O_DENOM>,
118                    "Rate::checked_add"
119                );
120
121                if Helpers::<NOM, DENOM, O_NOM, O_DENOM>::SAME_BASE {
122                    if let Some(raw) = self.raw.checked_add(other.raw) {
123                        Some(Self::from_raw(raw))
124                    } else {
125                        None
126                    }
127                } else {
128                    if let Some(lh) = other
129                        .raw
130                        .checked_mul(Helpers::<NOM, DENOM, O_NOM, O_DENOM>::LD_TIMES_RN as $i)
131                    {
132                        let raw = lh / Helpers::<NOM, DENOM, O_NOM, O_DENOM>::RD_TIMES_LN as $i;
133
134                        if let Some(raw) = self.raw.checked_add(raw) {
135                            Some(Self::from_raw(raw))
136                        } else {
137                            None
138                        }
139                    } else {
140                        None
141                    }
142                }
143            }

193            ///
194            /// Returns `None` if `other` is zero or the cross-base conversion overflows.

196            /// ```
197            /// # use fugit::*;
198            #[doc = concat!("let r1 = Rate::<", stringify!($i), ", 1, 1_000>::from_raw(10);")]

201            /// assert_eq!(r1.checked_rem(r2).unwrap().to_raw(), 1);

203            pub const fn checked_rem<const O_NOM: u64, const O_DENOM: u64>(
204                self,
205                other: Rate<$i, O_NOM, O_DENOM>,
206            ) -> Option<Self> {
207                assert_conversion_fits!(
208                    $i,
209                    Helpers::<NOM, DENOM, O_NOM, O_DENOM>,
210                    "Rate::checked_rem"
211                );
212
213                if other.raw == 0 {
214                    None
215                } else if Helpers::<NOM, DENOM, O_NOM, O_DENOM>::SAME_BASE {
216                    Some(Self::from_raw(self.raw % other.raw))
217                } else {
218                    if let Some(lh) = other
219                        .raw
220                        .checked_mul(Helpers::<NOM, DENOM, O_NOM, O_DENOM>::LD_TIMES_RN as $i)
221                    {
222                        let raw = lh / Helpers::<NOM, DENOM, O_NOM, O_DENOM>::RD_TIMES_LN as $i;
223
224                        if raw > 0 {
225                            Some(Self::from_raw(self.raw % raw))
226                        } else {
227                            None
228                        }
229                    } else {
230                        None
231                    }
232                }
233            }

247            /// Const partial comparison.
248            ///
249            /// Returns `None` if either side's raw value cannot be expressed in
250            /// the common base without overflowing the storage type.
251            ///
252            /// ```
253            /// # use fugit::*;
254            #[doc = concat!("let r1 = Rate::<", stringify!($i), ", 1, 1_00>::from_raw(1);")]
255            #[doc = concat!("let r2 = Rate::<", stringify!($i), ", 1, 1_000>::from_raw(1);")]
256            ///
257            /// assert_eq!(r1.const_partial_cmp(r2), Some(core::cmp::Ordering::Greater));
258            /// ```
259            #[inline]
260            pub const fn const_partial_cmp<const R_NOM: u64, const R_DENOM: u64>(
261                self,
262                other: Rate<$i, R_NOM, R_DENOM>
263            ) -> Option<Ordering> {
264                assert_conversion_fits!(
265                    $i,
266                    Helpers::<NOM, DENOM, R_NOM, R_DENOM>,
267                    "Rate::const_partial_cmp"
268                );
269
270                if Helpers::<NOM, DENOM, R_NOM, R_DENOM>::SAME_BASE {
271                    // If we are in the same base, comparison in trivial
272                    Some(Self::_const_cmp(self.raw, other.raw))
273                } else {
274                    let lh = self
275                        .raw
276                        .checked_mul(Helpers::<NOM, DENOM, R_NOM, R_DENOM>::RD_TIMES_LN as $i);
277                    let rh = other
278                        .raw
279                        .checked_mul(Helpers::<NOM, DENOM, R_NOM, R_DENOM>::LD_TIMES_RN as $i);
280
281                    if let (Some(lh), Some(rh)) = (lh, rh) {
282                        Some(Self::_const_cmp(lh, rh))
283                    } else {
284                        None
285                    }
286                }
287            }

289            /// Const equality check.
290            ///
291            /// Returns `false` (rather than panicking) if the cross-base
292            /// conversion overflows the storage type.
293            ///
294            /// ```
295            /// # use fugit::*;
296            #[doc = concat!("let r1 = Rate::<", stringify!($i), ", 1, 1_00>::from_raw(1);")]
297            #[doc = concat!("let r2 = Rate::<", stringify!($i), ", 1, 1_000>::from_raw(10);")]
298            ///
299            /// assert!(r1.const_eq(r2));
300            /// ```
301            #[inline]
302            pub const fn const_eq<const R_NOM: u64, const R_DENOM: u64>(
303                self,
304                other: Rate<$i, R_NOM, R_DENOM>
305            ) -> bool {
306                assert_conversion_fits!(
307                    $i,
308                    Helpers::<NOM, DENOM, R_NOM, R_DENOM>,
309                    "Rate::const_eq"
310                );
311
312                if Helpers::<NOM, DENOM, R_NOM, R_DENOM>::SAME_BASE {
313                    // If we are in the same base, comparison in trivial
314                    self.raw == other.raw
315                } else {
316                    let lh = self
317                        .raw
318                        .checked_mul(Helpers::<NOM, DENOM, R_NOM, R_DENOM>::RD_TIMES_LN as $i);
319                    let rh = other
320                        .raw
321                        .checked_mul(Helpers::<NOM, DENOM, R_NOM, R_DENOM>::LD_TIMES_RN as $i);
322
323                    if let (Some(lh), Some(rh)) = (lh, rh) {
324                        lh == rh
325                    } else {
326                        false
327                    }
328                }
329            }

331            /// Const try from, checking for overflow.
332            ///
333            /// ```
334            /// # use fugit::*;
335            #[doc = concat!("let r1 = Rate::<", stringify!($i), ", 1, 1_00>::from_raw(1);")]
336            #[doc = concat!("let r2 = Rate::<", stringify!($i), ", 1, 1_000>::const_try_from(r1);")]
337            ///
338            /// assert_eq!(r2.unwrap().to_raw(), 10);
339            /// ```
340            pub const fn const_try_from<const I_NOM: u64, const I_DENOM: u64>(
341                rate: Rate<$i, I_NOM, I_DENOM>,
342            ) -> Option<Self> {
343                if Helpers::<I_NOM, I_DENOM, NOM, DENOM>::SAME_BASE {
344                    Some(Self::from_raw(rate.raw))
345                } else {
346                    if let Some(lh) = (rate.raw as u64)
347                        .checked_mul(Helpers::<I_NOM, I_DENOM, NOM, DENOM>::RD_TIMES_LN)
348                    {
349                        let raw = div_round_nearest_u64(
350                            lh,
351                            Helpers::<I_NOM, I_DENOM, NOM, DENOM>::LD_TIMES_RN,
352                        );
353
354                        if raw <= <$i>::MAX as u64 {
355                            Some(Self::from_raw(raw as $i))
356                        } else {
357                            None
358                        }
359                    } else {
360                        None
361                    }
362                }
363            }

421                const {
