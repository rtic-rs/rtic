18unsafe fn waker_wake(p: *const ()) {
19    // The only thing we need from a waker is the function to call to pend the async
20    // dispatcher.
21    let f: fn() = unsafe { mem::transmute(p) };
22    f();
23}

30// Naming a task's future

32/// Binds the future type of an `async fn` as an associated type.
33///
34/// An `async fn`'s future type cannot be written down, so a `static` holding one cannot be
35/// declared directly. Naming it as `<F as ExecFn<Args>>::Fut` for the function's own type is
36/// enough to ask for its size and alignment in a `const`, which is what [`ExecutorHolder`]
37/// needs.
38pub trait ExecFn<Args>: Copy {
39    /// The future the function returns.
40    type Fut: Future + 'static;
41}
42
43macro_rules! exec_fn_impl {
44    ($($Tn:ident),*) => {
45        impl<F, Fut, $($Tn,)*> ExecFn<($($Tn,)*)> for F
46        where
47            F: Copy + FnOnce($($Tn,)*) -> Fut,
48            Fut: Future + 'static,
49        {
50            type Fut = Fut;
51        }
52    };
53}
54

108/// Storage for one [`AsyncTaskExecutor`], sized and aligned for a future that cannot be named.
109///
110/// Declared as bytes so the `static` needs no type parameter, and initialized by transmuting an
111/// [`exec_new`]. Both flags are false in that image, so this lands in `.bss` and costs no
112/// initializer.
113#[allow(private_bounds)]
114#[repr(C)]
115pub struct ExecutorHolder<const SIZE: usize, const ALIGN: usize>
116where
117    Align<ALIGN>: Alignment,
118{
119    data: UnsafeCell<[MaybeUninit<u8>; SIZE]>,
120    align: Align<ALIGN>,
121}
122
123unsafe impl<const SIZE: usize, const ALIGN: usize> Send for ExecutorHolder<SIZE, ALIGN> where
124    Align<ALIGN>: Alignment
125{
126}
127
128unsafe impl<const SIZE: usize, const ALIGN: usize> Sync for ExecutorHolder<SIZE, ALIGN> where
129    Align<ALIGN>: Alignment
130{
131}

133/// Reads back the executor an [`ExecutorHolder`] was initialized with.
134///
135/// # Safety
136///
137/// `holder` must have been initialized by transmuting [`exec_new`] applied to the same `_f`.
138#[allow(private_bounds)]
139#[inline(always)]
140pub unsafe fn exec_from_holder<F, Args, Fut, const SIZE: usize, const ALIGN: usize>(
141    _f: F,
142    holder: &'static ExecutorHolder<SIZE, ALIGN>,
143) -> &'static AsyncTaskExecutor<Fut>
144where
145    F: ExecFn<Args, Fut = Fut>,
146    Fut: Future + 'static,
147    Align<ALIGN>: Alignment,
148{
149    // Guard against a change to the `executor_decl` macro making it silently transmute the wrong bytes.
150    const {
151        assert!(SIZE == size_of::<AsyncTaskExecutor<Fut>>());
152        assert!(ALIGN == align_of::<AsyncTaskExecutor<Fut>>());
153    }
154
155    unsafe { &*holder.data.get().cast() }
156}

160pub struct Align<const N: usize>([<Self as Alignment>::Archetype; 0])
161where
162    Self: Alignment;

164trait Alignment {
165    /// A zero-sized type of particular alignment.
166    type Archetype: Copy + Eq + PartialEq + Send + Sync + Unpin;
167}

169macro_rules! aligns {
170    ($($AlignX:ident: $n:literal,)*) => {
171        $(
172            #[derive(Copy, Clone, Eq, PartialEq)]
173            #[repr(align($n))]
174            struct $AlignX {}
175
176            impl Alignment for Align<$n> {
177                type Archetype = $AlignX;
178            }
179        )*
180    };
181}

204/// Executor for an async task.
205pub struct AsyncTaskExecutor<F: Future + 'static> {
206    // `task` is protected by the `running` flag.
207    task: UnsafeCell<MaybeUninit<F>>,
208    running: AtomicBool,
209    pending: AtomicBool,
210}
211
212unsafe impl<F: Future + 'static> Sync for AsyncTaskExecutor<F> {}
213
214impl<F: Future + 'static> Default for AsyncTaskExecutor<F> {
215    fn default() -> Self {
216        Self::new()
217    }
218}
219
220impl<F: Future + 'static> AsyncTaskExecutor<F> {
221    /// Create a new executor.
222    #[inline(always)]
223    pub const fn new() -> Self {
224        Self {
225            task: UnsafeCell::new(MaybeUninit::uninit()),
226            running: AtomicBool::new(false),
227            pending: AtomicBool::new(false),
228        }
229    }
230
231    /// Check if there is an active task in the executor.
232    #[inline(always)]
233    pub fn is_running(&self) -> bool {
234        self.running.load(Ordering::Relaxed)
235    }
236
237    /// Checks if a waker has pended the executor and simultaneously clears the flag.
238    #[inline(always)]
239    fn check_and_clear_pending(&self) -> bool {
240        // Ordering::Acquire to enforce that update of task is visible to poll
241        self.pending
242            .compare_exchange(true, false, Ordering::Acquire, Ordering::Relaxed)
243            .is_ok()
244    }
245
246    // Used by wakers to indicate that the executor needs to run.
247    #[inline(always)]
248    pub fn set_pending(&self) {
249        self.pending.store(true, Ordering::Release);
250    }
251
252    /// Allocate the executor. To use with `spawn`.
253    #[inline(always)]
254    pub unsafe fn try_allocate(&self) -> bool {
255        // Try to reserve the executor for a future.
256        self.running
257            .compare_exchange(false, true, Ordering::AcqRel, Ordering::Relaxed)
258            .is_ok()
259    }
260
261    /// Spawn a future
262    #[inline(always)]
263    pub unsafe fn spawn(&self, future: F) {
264        // This unsafe is protected by `running` being false and the atomic setting it to true.
265        unsafe {
266            self.task.get().write(MaybeUninit::new(future));
267        }
268        self.set_pending();
269    }
270
271    #[inline(always)]
272    pub const fn waker(&self, wake: fn()) -> Waker {
273        unsafe { Waker::from_raw(RawWaker::new(wake as *const (), &WAKER_VTABLE)) }
274    }
275
276    /// Poll the future in the executor.
277    #[inline(always)]
278    pub fn poll(&self, wake: fn()) {
279        if self.is_running() && self.check_and_clear_pending() {
280            let waker = self.waker(wake);
281            let mut cx = Context::from_waker(&waker);
282            let future = unsafe { Pin::new_unchecked(&mut *(self.task.get() as *mut F)) };
283
284            match future.poll(&mut cx) {
285                Poll::Ready(_) => {
286                    self.running.store(false, Ordering::Release);
287                }
288                Poll::Pending => {}
289            }
290        }
291    }
292}
