1//! Compile-time lookups into the `stm32-metapac` metadata.
2//!
3//! The chip is selected by the chip feature enabled on the `stm32-metapac`
4//! dependency, so all lookups resolve to constants for that chip. A missing
5//! peripheral or RCC entry is a compile-time panic.
6
7use stm32_metapac::metadata::{
8    ir::{BitOffset, BlockItemInner},
9    Peripheral, PeripheralRccRegister, METADATA,
10};
11
12pub const NVIC_PRIO_BITS: u8 = match METADATA.nvic_priority_bits {
13    Some(bits) => bits,
14    None => panic!("stm32-metapac metadata does not provide `nvic_priority_bits`"),
15};

17// Errata ES0005 2.1.11: delay after an RCC peripheral clock enabling.
18const DSB_AFTER_ENABLE: bool = streq(METADATA.family, "STM32F2");
19
20/// RCC enable/reset bits for a timer, resolved at compile time.
21pub struct TimerRcc {
22    enable: RccBit,
23    reset: Option<RccBit>,
24}
25
26impl TimerRcc {
27    pub const fn lookup(timer: &str) -> Self {
28        let rcc = match &find_peripheral(timer).rcc {
29            Some(rcc) => rcc,
30            None => panic!("timer has no RCC metadata in stm32-metapac"),
31        };
32        Self {
33            enable: match &rcc.enable {
34                Some(enable) => RccBit::lookup(enable),
35                None => panic!("timer has no RCC enable metadata in stm32-metapac"),
36            },
37            reset: match &rcc.reset {
38                Some(reset) => Some(RccBit::lookup(reset)),
39                None => None,
40            },
41        }
42    }
43
44    pub fn enable(&self) {
45        self.enable.set(true);
46        if DSB_AFTER_ENABLE {
47            cortex_m::asm::dsb();
48        }
49    }
50
51    pub fn reset(&self) {
52        if let Some(reset) = &self.reset {
53            reset.set(true);
54            reset.set(false);
55        }
56    }
57}

59/// A single bit in an RCC register.
60#[derive(Clone, Copy)]
61struct RccBit {
62    addr: u32,
63    mask: u32,
64}
65
66impl RccBit {
67    /// Resolves an RCC (register, field) name pair to (address, mask) by
68    /// walking the RCC register block IR.
69    const fn lookup(bit: &PeripheralRccRegister) -> Self {
70        let rcc = find_peripheral("RCC");
71        let regs = match &rcc.registers {
72            Some(regs) => regs,
73            None => panic!("RCC has no registers in stm32-metapac metadata"),
74        };
75
76        let mut b = 0;
77        while b < regs.ir.blocks.len() {
78            let block = &regs.ir.blocks[b];
79            if !streq(block.name, regs.block) {
80                b += 1;
81                continue;
82            }
83
84            let mut i = 0;
85            while i < block.items.len() {
86                let item = &block.items[i];
87                if !streq(item.name, bit.register) {
88                    i += 1;
89                    continue;
90                }
91                assert!(
92                    item.array.is_none(),
93                    "RCC register arrays are not supported"
94                );
95                let fieldset = match &item.inner {
96                    BlockItemInner::Register(reg) => match reg.fieldset {
97                        Some(fieldset) => fieldset,
98                        None => panic!("RCC register has no fieldset"),
99                    },
100                    BlockItemInner::Block(_) => panic!("expected an RCC register, found a block"),
101                };
102                return Self {
103                    addr: rcc.address as u32 + item.byte_offset,
104                    mask: 1 << field_bit(regs.ir.fieldsets, fieldset, bit.field),
105                };
106            }
107            panic!("RCC register not found in stm32-metapac metadata");
108        }
109        panic!("RCC register block not found in stm32-metapac metadata");
110    }
111
112    /// Non-atomic read-modify-write, matching the plain `modify` the
113    /// build-script generated code used before.
114    fn set(&self, on: bool) {
115        let reg = self.addr as *mut u32;
116        unsafe {
117            let val = reg.read_volatile();
118            reg.write_volatile(if on {
119                val | self.mask
120            } else {
121                val & !self.mask
122            });
123        }
124    }
125}

127const fn field_bit(
128    fieldsets: &[stm32_metapac::metadata::ir::FieldSet],
129    fieldset: &str,
130    field: &str,
131) -> u32 {
132    let mut f = 0;
133    while f < fieldsets.len() {
134        if !streq(fieldsets[f].name, fieldset) {
135            f += 1;
136            continue;
137        }
138        let fields = fieldsets[f].fields;
139        let mut i = 0;
140        while i < fields.len() {
141            if streq(fields[i].name, field) {
142                match &fields[i].bit_offset {
143                    BitOffset::Regular(offset) => return offset.offset,
144                    BitOffset::Cursed(_) => panic!("cursed bit offset in RCC field"),
145                }
146            }
147            i += 1;
148        }
149        panic!("RCC field not found in stm32-metapac metadata");
150    }
151    panic!("RCC fieldset not found in stm32-metapac metadata");
152}
153
154const fn find_peripheral(name: &str) -> &'static Peripheral {
155    let mut i = 0;
156    while i < METADATA.peripherals.len() {
157        if streq(METADATA.peripherals[i].name, name) {
158            return &METADATA.peripherals[i];
159        }
160        i += 1;
161    }
162    panic!("peripheral not found in stm32-metapac metadata");
163}

165/// Case-insensitive ASCII compare; IR names are lowercase while RCC
166/// register/field names are uppercase.
167const fn streq(a: &str, b: &str) -> bool {
168    let (a, b) = (a.as_bytes(), b.as_bytes());
169    if a.len() != b.len() {
170        return false;
171    }
172    let mut i = 0;
173    while i < a.len() {
174        if !a[i].eq_ignore_ascii_case(&b[i]) {
175            return false;
176        }
177        i += 1;
178    }
179    true
180}
